\documentclass[tikz,border=3mm]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

\begin{document}
\begin{tikzpicture}
\begin{axis}[
    axis lines = left,
    xlabel = {$V$ (Volume)},
    ylabel = {$P$ (Pressure)},
    xtick = \empty,
    ytick = \empty,
    xmin = 0, xmax = 10,
    ymin = 0, ymax = 10,
    domain = 1.5:8.5,
    samples = 100,
    legend pos = north east,
    legend style = {draw=none, fill=none}
]

    % Isotherm T + dT
    \addplot[blue, thick] {9/x + 1} node[right, pos=0.9] {$T + dT$};
    
    % Isotherm T
    \addplot[cyan, thick] {6/x + 0.5} node[right, pos=0.95] {$T$};
    
    % Adiabatic expansion (left side of cycle)
    \addplot[red, dashed, domain=2:3.5] {13/(x^1.4) + 0.2};
    
    % Adiabatic compression (right side of cycle)
    \addplot[red, dashed, domain=4.5:7.2] {28/(x^1.4) - 0.5};
    
    % Shaded Carnot Cycle area (Approximation for Clapeyron physical derivation)
    \path[name path=top] (axis cs:2,5.5) plot[domain=2:4.5] (\x, {9/\x + 1});
    \path[name path=bottom] (axis cs:3.5,2.21) plot[domain=3.5:7.2] (\x, {6/\x + 0.5});
    
    % Core labels for the infinitesimal change
    \draw[black, <->, dashed] (axis cs:3, 2.5) -- (axis cs:3, 4) 
        node[midway, left] {$dP$};
    \draw[black, <->, dashed] (axis cs:3, 2.5) -- (axis cs:5.5, 2.5) 
        node[midway, below] {$dV$};

    % Title/Equation inside the graph
    \node[anchor=west] at (axis cs:5, 8) {\Large $\frac{dP}{dT} = \frac{L}{T \Delta V}$};

\end{axis}
\end{tikzpicture}
\end{document}
